    Evaluation¶

    Monte Carlo runs, the navigation-error definitions, and the summary figure of the command-line tool.

    ins_sim.evaluation.monte_carlo ¶

    Monte Carlo evaluation: many independent runs of the noisy IMU and the navigator.

    run_monte_carlo repeats the sensor-error draw and the strapdown navigation for a number of trials and collects every trial's estimates and attitude error. The spread across trials is the simulator's measure of navigation performance; percentile_envelope summarizes it as a position-error percentile over time. See the Monte Carlo evaluation theory page.

    run_monte_carlo ¶

    run_monte_carlo(truth: TruthTrajectory, spec: IMUSpec, n_trials: int, seed: int = 0, progress_callback: Callable[[int, int], None] | None = None, baro_aiding: bool = True) -> MonteCarloResult

    Runs n_trials independent noisy-IMU strapdown realizations.

    Each trial has its own random stream, spawned from seed, and each error source within a trial (initial alignment, IMU) draws from its own child stream. A trial's draws therefore do not depend on n_trials, and changing one error source's parameters leaves the other sources' realizations unchanged (common random numbers).

    def run_monte_carlo(truth: "TruthTrajectory", spec: IMUSpec,
                    n_trials: int, seed: int = 0,
                    progress_callback: Callable[[int, int], None] | None = None,
                    baro_aiding: bool = True) -> MonteCarloResult:
    """Runs n_trials independent noisy-IMU strapdown realizations.

    Each trial has its own random stream, spawned from `seed`, and each
    error source within a trial (initial alignment, IMU) draws from its own
    child stream. A trial's draws therefore do not depend on `n_trials`, and
    changing one error source's parameters leaves the other sources'
    realizations unchanged (common random numbers).

    Args:
        truth: Truth trajectory exposing t, tau_i, L_b, lambda_b, h_b,
            v_eb_n, C_b_n, omega_ib_b and f_ib_b.
        spec: IMU error-parameter set used to generate noisy samples.
        n_trials: Number of independent Monte Carlo trials.
        seed: Seed for the master random generator. Defaults to 0.
        progress_callback: Optional callable invoked as
            progress_callback(n_done, n_trials) after each trial
            completes. Defaults to None (no reporting).
        baro_aiding: When True the strapdown runs with the baro-damped
            vertical channel (truth height as the baro measurement);
            when False the vertical channel is free-inertial and
            diverges. Defaults to True.

    Returns:
        MonteCarloResult: estimates and attitude errors for every trial.

    Examples:
        >>> from ins_sim.evaluation.monte_carlo import run_monte_carlo
        >>> from ins_sim.sensors.imu import IMUSpec
        >>> from ins_sim.trajectory.kinematics import TruthTrajectory
        >>> from ins_sim.trajectory.spline import NEDSplinePath
        >>> path = NEDSplinePath([[0.0, 0.0, 0.0], [3000.0, 1000.0, -50.0]])
        >>> truth = TruthTrajectory.from_spline(path, speed=150.0, tau_i=0.1)
        >>> mc = run_monte_carlo(truth, IMUSpec(), n_trials=3, seed=1)
        >>> mc.est_L_b.shape == (3, len(truth.t))
        True
    """
    M = len(truth.t)
    est_r_lb_l     = np.zeros((n_trials, M, 3))
    est_euler_nb   = np.zeros((n_trials, M, 3))
    est_L_b        = np.zeros((n_trials, M))
    est_lambda_b   = np.zeros((n_trials, M))
    est_v_eb_n     = np.zeros((n_trials, M, 3))
    est_h_b        = np.zeros((n_trials, M))
    delta_psi_nb_n = np.zeros((n_trials, M, 3))

    # --8<-- [start:trial]
    trial_rngs = np.random.default_rng(seed).spawn(n_trials)
    C_b_n0 = truth.C_b_n[0]

    for i, trial_rng in enumerate(trial_rngs):
        rng_alignment, rng_imu = trial_rng.spawn(2)
        # Per-trial initial-alignment error delta_psi_nb^n: small NED tilts
        # plus an azimuth error, applied to the truth attitude (zero-mean
        # draws; exact truth initialization when the alignment stds are 0).
        delta_psi_nb_n0 = draw_initial_misalignment(spec, rng_alignment)
        est_C_b_n0 = Rot.from_rotvec(delta_psi_nb_n0) * C_b_n0
        init_state = NavState(
            truth.L_b[0], truth.lambda_b[0], truth.h_b[0],
            truth.v_eb_n[0].copy(),
            est_C_b_n0.as_quat(), # pyright: ignore[reportCallIssue]
        )
        meas_omega_ib_b, meas_f_ib_b = generate_imu_samples(truth, spec, rng_imu)
        nav = strapdown_navgrade(
            meas_omega_ib_b, meas_f_ib_b, init_state, truth.tau_i,
            meas_h_b=truth.h_b if baro_aiding else None)
        # --8<-- [end:trial]
        est_r_lb_l[i]   = nav.r_lb_l
        est_L_b[i]      = nav.L_b
        est_lambda_b[i] = nav.lambda_b
        est_h_b[i]      = nav.h_b
        est_v_eb_n[i]   = nav.v_eb_n
        est_C_b_n = Rot.from_quat(nav.q_b_n)
        delta_psi_nb_n[i] = attitude_error(est_C_b_n, truth.C_b_n)
        # as_euler('ZYX') → [psi, theta, phi]; reverse to [phi, theta, psi]
        est_euler_nb[i] = est_C_b_n.as_euler('ZYX')[:, ::-1]
        if progress_callback is not None:
            progress_callback(i + 1, n_trials)

    return MonteCarloResult(est_r_lb_l, est_euler_nb, est_L_b, est_lambda_b,
                            est_v_eb_n, est_h_b, delta_psi_nb_n)

    def percentile_envelope(delta_r_eb_n: np.ndarray, q: float = 95) -> np.ndarray:
    """Computes the per-time-step q-th percentile of 3D position error magnitude.

    Args:
        delta_r_eb_n: Position error per trial (see
            `ins_sim.evaluation.errors.position_error`), shape
            (n_trials, M, 3) [m].
        q: Percentile to compute, in [0, 100]. Defaults to 95.

    Returns:
        numpy.ndarray: q-th percentile of the 3D error magnitude across
            trials at each time step, shape (M,) [m].

    Examples:
        Five trials whose errors are 1 to 5 m at every sample:

        >>> import numpy as np
        >>> from ins_sim.evaluation.monte_carlo import percentile_envelope
        >>> delta_r = np.zeros((5, 2, 3))
        >>> delta_r[:, :, 0] = np.arange(1.0, 6.0)[:, None]
        >>> [float(x) for x in percentile_envelope(delta_r, q=50)]
        [3.0, 3.0]
    """
    # --8<-- [start:percentile]
    return np.percentile(np.linalg.norm(delta_r_eb_n, axis=2), q, axis=0)

    ins_sim.evaluation.errors ¶

    Navigation error definitions, in Groves (2013) notation.

    Every error is the estimate minus the truth. All three are resolved in the local NED axes at the true position, so attitude, velocity and position errors share one frame. They are also what the linearized error model (covariance analysis) propagates. The definitions are spelled out on Notation & conventions.

    attitude_error ¶

    attitude_error(est_C_b_n: Rotation, C_b_n: Rotation) -> ndarray

    Computes the attitude error delta_psi_nb^n.

    The small rotation that takes the true attitude to the estimated one: C_b^n(estimated) = exp([delta_psi_nb^n ^]) · C_b^n(true), so delta_psi_nb^n is the rotation vector of C_b^n(estimated)·C_n^b(true). Its components are the North tilt, the East tilt and the Down (heading) error, resolved in NED.

    Parameters:

    • est_C_b_n (Rotation) –

      Estimated body-to-NED attitude (scipy Rotation, single or stacked).

    • C_b_n (Rotation) –

      True body-to-NED attitude, with the same stacking.

    Returns:

    • ndarray –

      numpy.ndarray: delta_psi_nb^n, shape (3,) or (M, 3) [rad].

    Examples:

    A 1 mrad heading error appears on the Down component:

    >>> from scipy.spatial.transform import Rotation
>>> from ins_sim.evaluation.errors import attitude_error
>>> C_b_n = Rotation.from_euler("ZYX", [30.0, 2.0, 5.0], degrees=True)
>>> est_C_b_n = Rotation.from_rotvec([0.0, 0.0, 1e-3]) * C_b_n
>>> [round(float(x), 6) for x in attitude_error(est_C_b_n, C_b_n)]
[0.0, 0.0, 0.001]

    def attitude_error(est_C_b_n: Rotation, C_b_n: Rotation) -> np.ndarray:
    """Computes the attitude error delta_psi_nb^n.

    The small rotation that takes the true attitude to the estimated one:
    C_b^n(estimated) = exp([delta_psi_nb^n ^]) · C_b^n(true), so
    delta_psi_nb^n is the rotation vector of C_b^n(estimated)·C_n^b(true).
    Its components are the North tilt, the East tilt and the Down
    (heading) error, resolved in NED.

    Args:
        est_C_b_n: Estimated body-to-NED attitude (scipy Rotation, single
            or stacked).
        C_b_n: True body-to-NED attitude, with the same stacking.

    Returns:
        numpy.ndarray: delta_psi_nb^n, shape (3,) or (M, 3) [rad].

    Examples:
        A 1 mrad heading error appears on the Down component:

        >>> from scipy.spatial.transform import Rotation
        >>> from ins_sim.evaluation.errors import attitude_error
        >>> C_b_n = Rotation.from_euler("ZYX", [30.0, 2.0, 5.0], degrees=True)
        >>> est_C_b_n = Rotation.from_rotvec([0.0, 0.0, 1e-3]) * C_b_n
        >>> [round(float(x), 6) for x in attitude_error(est_C_b_n, C_b_n)]
        [0.0, 0.0, 0.001]
    """
    # --8<-- [start:attitude_error]
    return (est_C_b_n * C_b_n.inv()).as_rotvec()

    position_error ¶

    position_error(est_L_b: ArrayLike, est_lambda_b: ArrayLike, est_h_b: ArrayLike, L_b: ArrayLike, lambda_b: ArrayLike, h_b: ArrayLike) -> ndarray

    Computes the position error delta_r_eb^n, resolved in NED at the true position.

    The curvilinear errors (delta_L_b, delta_lambda_b, delta_h_b) are converted to metres along the local North, East and Down axes:

    delta_r_eb^n = [(R_N + h_b)·delta_L_b,
                (R_E + h_b)·cos(L_b)·delta_lambda_b,
                −delta_h_b]

    with the radii of curvature evaluated at the true latitude.

    Parameters:

    • est_L_b (ArrayLike) –

      Estimated latitude [rad], shape (..., M).

    • est_lambda_b (ArrayLike) –

      Estimated longitude [rad], shape (..., M).

    • est_h_b (ArrayLike) –

      Estimated height [m], shape (..., M).

    • L_b (ArrayLike) –

      True latitude [rad], shape (M,), broadcast against the estimates.

    • lambda_b (ArrayLike) –

      True longitude [rad], shape (M,).

    • h_b (ArrayLike) –

      True height [m], shape (M,).

    Returns:

    • ndarray –

      numpy.ndarray: delta_r_eb^n, shape (..., M, 3) [m].

    Examples:

    def position_error(est_L_b: npt.ArrayLike, est_lambda_b: npt.ArrayLike,
                   est_h_b: npt.ArrayLike, L_b: npt.ArrayLike,
                   lambda_b: npt.ArrayLike, h_b: npt.ArrayLike) -> np.ndarray:
    """Computes the position error delta_r_eb^n, resolved in NED at the true position.

    The curvilinear errors (delta_L_b, delta_lambda_b, delta_h_b) are
    converted to metres along the local North, East and Down axes:

        delta_r_eb^n = [(R_N + h_b)·delta_L_b,
                        (R_E + h_b)·cos(L_b)·delta_lambda_b,
                        −delta_h_b]

    with the radii of curvature evaluated at the true latitude.

    Args:
        est_L_b: Estimated latitude [rad], shape (..., M).
        est_lambda_b: Estimated longitude [rad], shape (..., M).
        est_h_b: Estimated height [m], shape (..., M).
        L_b: True latitude [rad], shape (M,), broadcast against the
            estimates.
        lambda_b: True longitude [rad], shape (M,).
        h_b: True height [m], shape (M,).

    Returns:
        numpy.ndarray: delta_r_eb^n, shape (..., M, 3) [m].

    Examples:
        A 1 µrad latitude error at 45° N is about 6.4 m North; a height
        estimate 10 m too high is a −10 m Down error:

        >>> import numpy as np
        >>> from ins_sim.evaluation.errors import position_error
        >>> L_b = np.deg2rad(45.0)
        >>> delta_r = position_error(L_b + 1e-6, 0.0, 10.0, L_b, 0.0, 0.0)
        >>> [round(float(x), 2) for x in delta_r]
        [6.37, 0.0, -10.0]
    """
    L_b = np.asarray(L_b)
    h_b = np.asarray(h_b)
    delta_L_b = np.asarray(est_L_b) - L_b
    # Wrap the longitude difference to (−π, π] so a track crossing the
    # antimeridian does not show a spurious 2π jump.
    delta_lambda_b = np.mod(np.asarray(est_lambda_b) - lambda_b + np.pi, 2.0 * np.pi) - np.pi
    delta_h_b = np.asarray(est_h_b) - h_b
    R_N, R_E = wgs84_radii(L_b)
    # --8<-- [start:position_error]
    return np.stack([
        (R_N + h_b) * delta_L_b,
        (R_E + h_b) * np.cos(L_b) * delta_lambda_b,
        -delta_h_b,
    ], axis=-1)

        r95: Radial error envelope at each time step, shape (M,), in
            the same units as r_lb_l.
        color: Surface color. Defaults to "crimson".
        alpha: Surface opacity. Defaults to 0.20.
        n_circle: Number of points approximating the tube's circular
            cross-section. Defaults to 24.

    Returns:
        None: Draws directly onto `ax`.
    """
    T = np.gradient(r_lb_l, axis=0)
    T = T / np.linalg.norm(T, axis=1, keepdims=True)

    seed = np.array([0.0, 0.0, 1.0])
    if abs(T[0] @ seed) > 0.95:
        seed = np.array([0.0, 1.0, 0.0])

    N1 = np.zeros_like(T); N2 = np.zeros_like(T)
    n  = seed - T[0] * (T[0] @ seed)
    N1[0] = n / np.linalg.norm(n)
    N2[0] = np.cross(T[0], N1[0])
    for k in range(1, len(T)):
        proj = N1[k-1] - T[k] * (T[k] @ N1[k-1])
        nn   = np.linalg.norm(proj)
        N1[k] = proj / nn if nn > 1e-10 else N1[k-1]
        N2[k] = np.cross(T[k], N1[k])

    th = np.linspace(0.0, 2*np.pi, n_circle)
    c, s = np.cos(th), np.sin(th)
    tube = (r_lb_l[:, None, :]
            + r95[:, None, None] * (c[None, :, None] * N1[:, None, :]
                                  + s[None, :, None] * N2[:, None, :]))

    X = tube[..., 1]      # East
    Y = tube[..., 0]      # North
    Z = -tube[..., 2]     # Up
    ax.plot_surface(X, Y, Z, color=color, alpha=alpha,
                    edgecolor='none', linewidth=0)

    def build_summary_figure(truth: "TruthTrajectory", mc: "MonteCarloResult",
                         r95: np.ndarray) -> plt.Figure:
    """Builds the 2x3 Monte Carlo summary figure.

    Figure panels: 3D ground track with swept 95th-pct error tube,
    3D position-error envelope, CEP, and the three components of the
    attitude error delta_psi_nb^n. Does not call plt.show() -- the
    caller decides.

    Note:
        Applies sns.set_theme(...) as a side effect (global rcParams
        mutation), matching the original script's behavior.

    Args:
        truth: Truth trajectory exposing r_lb_l, t, L_b, lambda_b and h_b.
        mc: MonteCarloResult from run_monte_carlo.
        r95: 95th-percentile 3D position-error envelope, shape (M,) [m].

    Returns:
        matplotlib.figure.Figure: The assembled summary figure.
    """
    sns.set_theme(
        style='whitegrid',
        palette='deep',
        font_scale=1.05,
        rc={
            'figure.facecolor': '#f5f5f5',
            'axes.facecolor':   'white',
            'grid.color':       '#dddddd',
            'axes.edgecolor':   '#aaaaaa',
            'text.color':       '#111111',
            'axes.labelcolor':  '#111111',
            'xtick.color':      '#333333',
            'ytick.color':      '#333333',
        },
    )

    n_trials = len(mc.est_r_lb_l)
    P   = truth.r_lb_l
    t_m = truth.t / 60.0   # minutes for x-axis
    delta_r_eb_n = position_error(mc.est_L_b, mc.est_lambda_b, mc.est_h_b,
                                  truth.L_b, truth.lambda_b, truth.h_b)

    step = max(1, len(P) // 4000)
    fig  = plt.figure(figsize=(22, 12), layout='constrained')
    gs   = fig.add_gridspec(2, 3)

    ax  = fig.add_subplot(gs[0, 0], projection='3d')
    ax2 = fig.add_subplot(gs[0, 1])
    ax3 = fig.add_subplot(gs[0, 2])
    ax4 = fig.add_subplot(gs[1, 0])   # North tilt
    ax5 = fig.add_subplot(gs[1, 1])   # East tilt
    ax6 = fig.add_subplot(gs[1, 2])   # Heading

    # ---- 3-D ground track + 95th-pct error tube ---------------------------
    # Isotropic km units (not the nm/kft mix used elsewhere) so the tube's
    # circular cross-section and tangent direction stay geometrically
    # correct: plot_error_tube offsets by a scalar radius (r95, in meters)
    # along normals derived from the track's own tangent, so all three axes
    # must share one linear scale factor.
    P_km = P / 1000.0
    r95_km = r95 / 1000.0
    ax.plot(P_km[::step, 1], P_km[::step, 0], -P_km[::step, 2],
            'k-', lw=1.5, label='Truth')
    for i in range(min(6, n_trials)):
        Q_km = mc.est_r_lb_l[i] / 1000.0
        ax.plot(Q_km[::step, 1], Q_km[::step, 0], -Q_km[::step, 2],
                '-', color='steelblue', alpha=0.35, lw=0.5)
    plot_error_tube(ax, P_km, r95_km)
    ax.set_xlabel('East (km)')
    ax.set_ylabel('North (km)')
    ax.set_zlabel('Up (km)')
    ax.set_title(f'{n_trials}-trial Monte Carlo')
    ax.legend(loc='upper left')

    # ---- 3-D position error vs time ---------------------------------------
    err_all = np.linalg.norm(delta_r_eb_n, axis=2)
    ax2.plot(t_m, err_all.T, color='steelblue', alpha=0.20, lw=0.5)
    ax2.plot(t_m, r95, color='crimson', lw=2.0, label='95th pct')
    ax2.fill_between(t_m, 0, r95, color='crimson', alpha=0.15)
    ax2.set_xlabel('Time (min)')
    ax2.set_ylabel('3-D position error (m)')
    ax2.set_title('Position error envelope')
    ax2.legend()

    # ---- CEP vs time + linear fit -----------------------------------------
    horiz_err = np.linalg.norm(delta_r_eb_n[:, :, :2], axis=2)
    cep = np.percentile(horiz_err, 50, axis=0)

    mask60 = t_m <= 60.0
    coeffs = np.polyfit(t_m[mask60], cep[mask60], 1)
    cep_fit = np.polyval(coeffs, t_m)

    ax3.plot(t_m, horiz_err.T, color='steelblue', alpha=0.15, lw=0.5)
    ax3.plot(t_m, cep, color='navy', lw=2.0, label='CEP (50th pct)')
    ax3.fill_between(t_m, 0, cep, color='steelblue', alpha=0.18)
    ax3.plot(t_m, cep_fit, '--', color='darkorange', lw=1.8,
             label=f'Linear fit  {coeffs[0]*0.032397408:.2f} NM/hr')
    ax3.axvline(60.0, color='gray', lw=0.8, linestyle=':')
    ax3.set_xlabel('Time (min)')
    ax3.set_ylabel('Horizontal error (m)')
    ax3.set_title('CEP along path')
    ax3.legend()

    # ---- Attitude error delta_psi_nb^n ------------------------------------
    delta_psi_deg = np.rad2deg(mc.delta_psi_nb_n)
    p95_abs = np.percentile(np.abs(delta_psi_deg), 95, axis=0)

    for ax_att, col, title in [
        (ax4, 0, 'North tilt error'),
        (ax5, 1, 'East tilt error'),
        (ax6, 2, 'Heading error'),
    ]:
        ax_att.plot(t_m, delta_psi_deg[:, :, col].T, color='steelblue',
                    alpha=0.20, lw=0.5)
        ax_att.fill_between(t_m, -p95_abs[:, col], p95_abs[:, col],
                            color='crimson', alpha=0.20,
                            label='95th pct |error|')
        ax_att.axhline(0.0, color='gray', lw=0.8)
        ax_att.set_xlabel('Time (min)')
        ax_att.set_ylabel(f'{title} (deg)')
        ax_att.set_title(title)
        ax_att.legend(fontsize=8)

    return fig
